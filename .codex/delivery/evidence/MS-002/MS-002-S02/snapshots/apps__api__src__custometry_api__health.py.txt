"""Health contracts and bounded PostgreSQL/schema/storage readiness adapters."""

from dataclasses import dataclass
import os
import secrets
from typing import Protocol

import psycopg

from custometry_api.config import Settings
from custometry_api.installation import InstallationComponents, InstallationStatus

SUPPORTED_HEAD = "0009_notifications"


class ReadinessProbe(Protocol):
    """Port used by the existing API health boundary."""

    def is_ready(self) -> bool: ...


def storage_ready(settings: Settings) -> bool:
    """Exercise create/write/fsync/unlink at the canonical root without making it."""
    root = settings.analytics_artifact_root
    directory = None
    name = ".readiness-" + secrets.token_hex(16)
    created = False
    try:
        if not root.is_absolute() or root.resolve(strict=True) != root:
            return False
        directory = os.open(root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        fd = os.open(
            name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600, dir_fd=directory
        )
        created = True
        try:
            os.write(fd, b"ready\n")
            os.fsync(fd)
        finally:
            os.close(fd)
        os.unlink(name, dir_fd=directory)
        created = False
        os.fsync(directory)
        return True
    except (OSError, ValueError, RuntimeError):
        return False
    finally:
        if directory is not None:
            if created:
                try:
                    os.unlink(name, dir_fd=directory)
                except OSError:
                    pass
            os.close(directory)


@dataclass(frozen=True, slots=True)
class PostgreSQLReadinessProbe:
    """Inspect operational state only; never migrate or query domain tables."""

    settings: Settings

    def inspect(self) -> InstallationStatus:
        database = False
        schema = False
        try:
            with psycopg.connect(
                host=self.settings.database_host,
                port=self.settings.database_port,
                dbname=self.settings.database_name,
                user=self.settings.database_user,
                password=self.settings.read_database_password(),
                connect_timeout=self.settings.database_connect_timeout_seconds,
                options="-c statement_timeout=2000 -c default_transaction_read_only=on",
            ) as connection:
                with connection.cursor() as cursor:
                    cursor.execute("SELECT 1")
                    database = cursor.fetchone() == (1,)
                    cursor.execute("SELECT version_num FROM public.alembic_version")
                    schema = cursor.fetchall() == [(SUPPORTED_HEAD,)]
        except (OSError, ValueError, psycopg.Error):
            pass
        storage = storage_ready(self.settings)
        code = (
            "DATABASE_UNAVAILABLE"
            if not database
            else "SCHEMA_INCOMPATIBLE"
            if not schema
            else "STORAGE_UNAVAILABLE"
            if not storage
            else None
        )
        return InstallationStatus(
            state="ready_for_bootstrap" if code is None else "not_ready",
            version=self.settings.version,
            components=InstallationComponents(
                database="ready" if database else "not_ready",
                schema="ready" if schema else "not_ready",
                storage="ready" if storage else "not_ready",
            ),
            code=code,
        )

    def is_ready(self) -> bool:
        return self.inspect().state == "ready_for_bootstrap"
